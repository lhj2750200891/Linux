## 2. 搭建交叉编译环境与烧录

```bash
# 下载压缩包
mkdir -p prebuilts/gcc/linux-x86/arm/
cd prebuilts/gcc/linux-x86/arm/
wget https://armkeil.blob.core.windows.net/developer/Files/downloads/gnu/13.2.rel1/binrel/arm-gnu-toolchain-13.2.rel1-x86_64-arm-none-eabi.tar.xz
tar xvf arm-gnu-toolchain-13.2.rel1-x86_64-arm-none-eabi.tar.xz
cd /media/lhj/windows/code/arm64/rockchip/rk2118_sdk/bsp/rockchip/rk2118
vim build.sh   # 最前面增加下面的内容
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
export RTT_EXEC_PATH="${SCRIPT_DIR}/../../../prebuilts/gcc/linux-x86/arm/arm-gnu-toolchain-13.2.Rel1-x86_64-arm-none-eabi/bin"

# 安装编译环境,编译的时候缺少什么安装什么即可
sudo apt install scons
sudo apt install clang-format 
sudo apt install astyle 
sudo apt install build-essential
sudo apt install dosfstools 
sudo apt install fusefat 
sudo apt install python3 
sudo apt install python3-pip 
sudo apt install lzma   # > 9.22

# 编译,编译弹出来menuconfig的时候,exit推出即可:自动完成编 cpu1(rtt1) → recovery → cpu0(rtt0) → 打 root.img → 打包,产物Image/update.img
cd /media/lhj/windows/code/arm64/rockchip/rk2118_sdk/bsp/rockchip/rk2118
./build.sh board/evb1_v20/package-file    (我的板子是v21的,如果是v1系列,用下面的指令)
./build.sh board/evb/package-file
# 构建成功输出如下
Add CRC...
Make firmware OK!
------ OK ------
********rkImageMaker ver 2.31********
Generating new image, please wait...
Writing head info...
Writing boot file...
Writing firmware...
Generating MD5 data...
MD5 data generated successfully!
New image generated successfully!

# 烧录
# 如果没有固件,则自动进入maskrom
# 如果进入系统,使用 maskrom + reset 按键进入烧录模式
# 软件进入: reboot loader
# linux下的烧录: typec 连接到 usb 2.0 device接口(靠近debug)
reboot loader
cd /media/lhj/windows/code/arm64/rockchip/rk2118_sdk/bsp/rockchip/rk2118
../tools/upgrade/upgrade_tool uf ./Image/update.img
# 如果失败,下 db loader后重新烧
../tools/upgrade/upgrade_tool db ./rkbin/rk2118_db_loader.bin
../tools/upgrade/upgrade_tool uf ./Image/update.img

# 单独烧录
# 例 1: 只烧 cpu0(改了应用/驱动,最常用)
sudo ../tools/upgrade/upgrade_tool wl 0x2200 ./Image/rtt0.img
# 例 2: 只烧 DSP1
sudo ../tools/upgrade/upgrade_tool wl 0xb200 ./Image/dsp1.img
# wl 的地址直接用 parameter.txt 里的 sector 值(工具内部 ×512)。烧的文件用 Image/ 下带 RESC 头的 .img

# 日常开发只需要记住一条：改了 MCU 代码就烧 rtt0.img 到 cpu0_os_a:

sudo ../tools/upgrade/upgrade_tool db ./rkbin/rk2118_db_loader.bin   # 进 maskrom 后先下 loader
sudo ../tools/upgrade/upgrade_tool wl 0x2200 ./Image/rtt0.img

# EVB V20(NAND, A/B 双分区)的完整分区表和固件对照关系如下。
```

## 3. 验证接口功能

```bash
# 网口测试,ping局域网设备即可
ifconfig # 查看网卡: network interface device: e0 (Default),这里是e0,网卡是UP状态即可
ping -I e0 172.16.10.222   # 默认是4包
ping -c 10 -I e0 172.16.10.222    # 连发 10 包
# tftp传输文件到sd卡   (需要pc上安装tftp: sudo apt install tftp)
cd bsp/rockchip/rk2118
vim board/evb1_v20/defconfig   # 打开 TFTP 和 TELNET: CONFIG_RT_NETUTILS_TFTP=y, CONFIG_RT_NETUTILS_TELNET=y
./build.sh board/evb1_v20/package-file
# 烧写(整包 uf 或单分区 wl 0x2200 Image/rtt0.img)
# 烧录完成以后,传输文件测试
tftp -s    # 板端启动服务,输出:tftp server start!
tftp 172.16.11.44    # pc 端连接板子,推出用quit


# usb 2.0 host(u盘接口): rtthread 只能认识fat16/32的盘

# sd卡测试,插入sd卡会输出如下的内容
msh />card connected
rk_mmc_request,get response returns -2, cmd: 5
[I/SDIO] SD card capacity 30253056 KB.
found part[0], begin: 4194304, size: 28.868GB
# 验证 mount 查看挂载点
msh />mount
filesystem  device  mountpoint
----------  ------  ----------
elm         root_a  /
elm         userda  /data
elm         sd0     /sdcard
cd /sdcard
# 播放wav音频
key_test 0xdb   # 切换模式,,或者使用mode按键切换,每次查看display mode
# [display_app: display_mode: 414]display_mode earc
# 一共有mode: usb,bt,earc,hdmi0,hdmi1,hdmi2,opt,aux,sdcard
# sdcard 模式: 会自动开始播放 sd卡中的音乐
# bt 模式: 手机连接 rk_soundar_a6f3 就可以播放测试
# usb 模式: 需要拔出typec接线,选择usb模式的时候,就会自动播放u盘里面的音乐
# earc 模式: hdmi 回传,电视带有earc时,声音回传到板卡的声卡,接hdmi rx
# opt 模式: 光纤输入模式,spdifrx接口,光纤数字音频
# aux 模式: 模拟音频输入
# hdmi0/1/2 模式: hdmi0 采样音频,外部音频接入hdmi rx,模式为hdmi0 即可播放声音
tracklist_info sdcard summary    # 扫描sd卡中的结果
player_select_dir /sdcard   # 选择播放目录
tracklist_info sdcard tracks   # 列出播放列表

```



## 4. 新增自己的板子

```bash
# 进入board目录
cd /media/lhj/windows/code/arm64/rockchip/rk2118_sdk/bsp/rockchip/rk2118/board
cp -r evb1_v20 firefly_rk2118g_evb
vim Kconfig  # choice 字段下增加自己的板子的配置,如下
choice
    prompt "Board Type"
    default RT_USING_BOARD_FPGA

    config RT_USING_BOARD_EVB1_V20
        bool "Enable evb1 v20 board"
        select RT_USING_DDR
  
    config RT_USING_BOARD_FIREFLY_RK2118G_EVB
        bool "Enable firefly rk2118g evb board"
        select RT_USING_DDR
endchoice
vim firefly_rk2118g_evb/SConscript    
# 在depend 字段中替换为: RT_USING_BOARD_FIREFLY_RK2118G_EVB

```
